export const famNames = Object.keys(primitives);

export function primHex(ref) {
  if (!ref) return null;
  const m = String(ref).match(/#[0-9a-fA-F]{6}/);
  if (m) return m[0];
  const low = String(ref).toLowerCase();
  if (low.startsWith("white")) return "#ffffff";
  for (const fam of famNames) {
    if (low.startsWith(fam + "-")) {
      const step = low.slice(fam.length + 1).split(" ")[0];
      const hit = (primitives[fam] || []).find((s) => s.step === step);
      if (hit) return hit.hex;
    }
  }
  return null;
}

export function Swatch({ hex, size = 40 }) {
  if (!hex) return null;
  return (
    <span
      aria-hidden
      dir="ltr"
      style={{
        display: "inline-block",
        width: size,
        height: size,
        minWidth: size,
        borderRadius: 12,
        background: hex,
        border: "1px solid var(--color-border)",
        boxShadow: "var(--shadow-card)",
        verticalAlign: "middle",
      }}
    />
  );
}

export function TokenTable({ rows }) {
  return (
    <div dir="ltr" style={{ display: "flex", flexDirection: "column", gap: 8, marginTop: 12 }}>
      {rows.map((t) => {
        const lh = primHex(t.light);
        const dh = primHex(t.dark);
        return (
          <div
            key={t.token}
            style={{
              display: "flex",
              alignItems: "center",
              gap: 12,
              padding: "10px 14px",
              borderRadius: 12,
              background: "var(--color-surface)",
              border: "1px solid var(--color-border)",
              fontSize: 13,
            }}
          >
            <span style={{ display: "inline-flex", gap: 4 }}>
              <Swatch hex={lh} size={32} />
              <Swatch hex={dh} size={32} />
            </span>
            <span style={{ minWidth: 0 }}>
              <strong style={{ color: "var(--color-foreground)" }}>
                <code>{t.token}</code>
              </strong>{" "}
              <span style={{ color: "var(--color-muted-foreground)" }}>
                {t.light} → {t.dark}
              </span>
              <br />
              <span style={{ color: "var(--color-muted-foreground)" }}>{t.usage}</span>{" "}
              <code style={{ color: "var(--color-foreground-faint)", fontSize: 12 }}>{t.tailwind}</code>
            </span>
          </div>
        );
      })}
    </div>
  );
}

### Accessibility & States

### Semantic Feedback

رنگ پس‌زمینه، رنگ متن و رنگ Border هر Semantic State باید همیشه با هم تعریف و استفاده شوند و نباید جدا از هم انتخاب شوند.

هر Feedback Box باید علاوه بر رنگ، یک Icon مناسب و یک `role` مشخص داشته باشد (`status` یا `alert`). مفهوم وضعیت نباید فقط از طریق رنگ منتقل شود.

**Info — خاکستری مایل به سبز، بدون آبی اشباع‌شده**

یک پیام خنثی و آرام با ته‌رنگ سرد.
نباید شبیه آبی استاندارد و اشباع‌شده باشد.

**Success — تأییدی آرام، بدون هیجان اضافی**

نشانه‌ای از پیشرفت و نتیجه‌ی مثبت، اما بدون حس جشن یا اغراق.
از یک check ملایم و حس پیشرفت آرام استفاده کن.

**Warning — یک تلنگر گرم، نه تذکر یا سرزنش**

لحن باید دوستانه و انسانی باشد؛ چیزی شبیه این مفهوم:
«هر روز می‌تواند یک شروع تازه باشد. از همین‌جا ادامه بدهیم.»

**Danger — فقط برای خطاهای واقعی**

پیام باید آرام، مشخص و کاربردی باشد و قدم بعدی را روشن کند.
از قرمز هشداردهنده و بیش‌ازحد تهاجمی استفاده نکن.

### WCAG 2.2 Contrast Validation

تمام نسبت‌های کنتراست زیر بر اساس WCAG 2.2 و با استفاده از relative luminance محاسبه شده‌اند؛ این اعداد تخمینی نیستند.

اهداف کنتراست:

- متن معمولی: حداقل **4.5:1**
- متن بزرگ: حداقل **3:1**
- مرزها و Boundaryهای ضروری: حداقل **3:1**

### نمونه‌های نادرست — Don't

این موارد نباید در UI استفاده شوند:

- متن کوچک سفید روی `sage-500`؛ برای مثال «رزرو کن» با کنتراست 3.11:1 ❌
- Border ضروری Input با رنگ بسیار کم‌رنگ Cream و نسبت 1.27:1 ❌
- نمایش خطا فقط با تغییر رنگ، بدون Icon و بدون `role` ❌

### نمونه‌های درست — Do

- دکمه‌ی `action-primary` با متن «رزرو کن» و کنتراست 6.09:1 ✅
- Inputها با `border-strong` و کنتراست 5.46:1 نسبت به زمینه‌ی مربوطه ✅
- خطاها همراه با Icon، `role="alert"` و متن با کنتراست قوی نمایش داده شوند ✅
- تمام Linkها همیشه underline داشته باشند تا قابل تشخیص بودن آن‌ها فقط به رنگ وابسته نباشد ✅

### قانون اصلی رنگ

از موارد زیر استفاده نکن:

- مشکی خالص
- سفید خالص
- رنگ‌های Neon
- رنگ‌های Acid
- Accentهای بیش‌ازحد تند
- آبی‌های اشباع‌شده
- بنفش‌های رایج SaaS
- قرمزهای هشداردهنده و Alarm-like

در نهایت، با استفاده از **Mood toolbar ↑** تمام نسبت‌های بالا را در حالت Dark نیز بررسی کن و مطمئن شو که جفت‌رنگ‌های مربوط به Dark Mode همچنان مطابق همین اصول عمل می‌کنند.
