# Icons

The shared icon set: one stroke family (1.75px, 24 viewBox, rounded caps)
so every icon on screen reads as one visual language. Add new icons in
`design-system/icons.tsx` rather than one-off inline SVGs in features.
`StarIcon` takes an optional `filled` prop; the rest take standard SVG props.

export const iconNames = Object.keys(IconSet).filter((k) => k.endsWith("Icon"));

// Audited 2026-10-05 against product imports: these 36 ship on real screens.
export const usedIcons = [
  "AccompaniedLineIcon", "ArrowLeftIcon", "ArrowRightIcon", "AwardIcon",
  "BookIcon", "BoltIcon", "BuildingIcon", "CalendarIcon", "CameraIcon",
  "ChatIcon", "CheckIcon", "ChevronDownIcon", "ClockIcon", "CoinIcon",
  "CompassIcon", "EditIcon", "FemaleIcon", "HomeIcon", "InstagramIcon",
  "LockIcon", "LogOutIcon", "MaleIcon", "MapPinIcon", "MenuIcon",
  "PhoneIcon", "ProfileIcon", "ProgressIcon", "SearchIcon", "SendIcon",
  "ShieldCheckIcon", "SparkleIcon", "StarIcon", "UploadIcon", "UsersIcon",
  "VideoIcon", "XIcon",
];
// The rest below are reusable reserve — generic icons kept for near-term
// needs, not currently consumed by any screen. Do not invent usages for them.
export const reserveIcons = iconNames.filter((n) => !usedIcons.includes(n));

export function IconTile({ name }) {
  const Cmp = IconSet[name];
  return (
    <div
      style={{
        display: "flex",
        flexDirection: "column",
        alignItems: "center",
        gap: 6,
        width: 104,
        padding: "12px 8px",
        borderRadius: 12,
        background: "var(--color-surface)",
        border: "1px solid var(--color-border)",
        color: "var(--color-foreground)",
      }}
    >
      <Cmp style={{ width: 24, height: 24 }} />
      <code style={{ fontSize: 10, textAlign: "center", overflowWrap: "anywhere" }}>{name}</code>
    </div>
  );
}
